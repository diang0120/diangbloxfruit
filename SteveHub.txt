local parent
if type(gethui) == "function" then
    local ok, result = pcall(gethui)
    if ok then parent = result end
end
if not parent then
    local ok, result = pcall(function() return game:GetService("CoreGui") end)
    if ok then parent = result end
end

    local ok = pcall(function() gui.Parent = parent end)
    if not ok then gui.Parent = player:WaitForChild("PlayerGui") end

    gui.Parent = player:WaitForChild("PlayerGui")
end

local function makeButton(name, top, primary)
    local button = Instance.new("TextButton")
    button.Name = name .. "Button"
    button.Position = UDim2.new(0, 20, 0, top)
    button.Size = UDim2.new(1, -40, 0, 51)
    button.BackgroundColor3 = primary and Color3.fromRGB(237, 190, 69) or Color3.fromRGB(42, 39, 30)
    button.BorderSizePixel = 0
    button.AutoButtonColor = true
    button.Font = Enum.Font.SourceSansBold
    button.Text = name:upper()
    button.TextColor3 = primary and Color3.fromRGB(20, 19, 16) or Color3.fromRGB(245, 211, 126)
    button.TextSize = 20
    button.Parent = card

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 12)
    corner.Parent = button
    return button
end

local busy = false
local function launch(which)
    if busy then return end
    busy = true
    status.Text = "Running " .. which .. "..."
    status.TextColor3 = Color3.fromRGB(242, 198, 75)

    task.spawn(function()
        local ok, err = pcall(function()
            if which == "Main" then
                loadstring(game:HttpGet("https://raw.githubusercontent.com/diang0120/diangbloxfruit/refs/heads/main/diangbloxfruit.lua"))()
            else
                loadstring(game:HttpGet("https://raw.githubusercontent.com/diang0120/diangbloxfruit/refs/heads/main/SteveKaitun.txt"))()
            end
        end)
        if not gui.Parent then return end
        if ok then
            status.Text = "Running " .. which
            task.delay(0.8, function()
                if gui.Parent then gui:Destroy() end
            end)
        else
            status.Text = "Error " .. which .. ": " .. tostring(err):sub(1, 160)
            status.TextColor3 = Color3.fromRGB(248, 134, 117)
            busy = false
        end
    end)
end

mainButton.MouseButton1Click:Connect(function() launch("Main") end)
kaitunButton.MouseButton1Click:Connect(function() launch("Kaitun") end)
